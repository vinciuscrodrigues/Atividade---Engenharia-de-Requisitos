4. Papéis na inspeção formal de Fagan:
Autor: produz o documento, esclarece dúvidas e corrige os defeitos identificados após a inspeção.
Moderador: conduz a sessão, controla o processo e mantém o foco na identificação de defeitos, evitando debates sobre soluções.
Inspetor/Revisor: examina previamente a especificação e identifica ambiguidades, omissões, contradições e outros problemas.
Relator: registra os defeitos encontrados, sua localização, classificação e severidade.

# Instalar dependencias: pip install cryptography beautifulsoup4
# Coloque este script na mesma pasta do HTML da Lista 12.
from pathlib import Path
from bs4 import BeautifulSoup
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from cryptography.exceptions import InvalidTag
import base64
import hashlib

arquivo = Path("LISTA_EXERCICIOS_SEMANA_12_INTERATIVA.html")
soup = BeautifulSoup(arquivo.read_text(encoding="utf-8"), "html.parser")
payload = soup.select_one("#encrypted-payload")
iv = base64.b64decode(payload["data-iv"])
ciphertext = base64.b64decode(payload["data-ct"])
salt = "unicesumar-er-2026"
wordlist = ["sommerville", "fagan", "boehm", "pressman", "wireframe",
            "prototipo", "smoketest", "validacao", "uat"]

for termo in wordlist:
    chave = hashlib.sha256((termo.strip().lower() + ":" + salt).encode("utf-8")).digest()
    try:
        AESGCM(chave).decrypt(iv, ciphertext, None)
    except InvalidTag:
        continue
    print("Palavra-chave encontrada:", termo)
    print("Descriptografia validada com sucesso.")
    break
else:
    print("Nenhuma palavra da lista conseguiu desbloquear o cofre.")
